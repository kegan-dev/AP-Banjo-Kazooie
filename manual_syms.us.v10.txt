/* BKBT (see BKBT.md). Nothing below moves anything: three absolute symbols and assertions. */

/* The number of the dual-boot contract this ROM implements (BKBT.md, "The frame hook"; the Tooie side's
   copy is DUALBOOT_CONTRACT in hack/bt/n64/src/dualboot.h of the BKBT repository). Not an address: a
   symbol, so that the map carries the number for the tools that pair this ROM with a launcher and a
   payload. The C code takes it from here too, as (u32) bkbt_contract (include/bkbt.h), so this line is
   the only place that says it. It is the low byte of the launch token: keep it below 0x100. */
bkbt_contract = 2;
ASSERT(bkbt_contract < 0x100, "BKBT: bkbt_contract does not fit the low byte of the launch token");

/* Where this game's 0x4A0 bytes of save lie in the SRAM of the dual-boot image (32 KiB at cartridge address
   0x08000000; src/core1/eeprom.c). An offset into the SRAM, not an address: a symbol like bkbt_contract,
   so that the map carries the number for the tools, and C takes it from here as (u32) bkbt_sram_offset
   (include/bkbt.h). 0x7000 is the border the two games share: below it the SRAM is the Tooie hack's, with
   room for its save to grow (2884 bytes today), and from it on this game's, with room behind the 0x4A0
   bytes for later BKBT data (four game records of 0x120 bytes and the global record of 0x20, since the
   notes' build; 0x200 bytes with retail's 0x78-byte records, 0x260 with the 0x90-byte ones, 0x2A0 with
   the Jinjos' 0xA0-byte ones). The
   Tooie side's copy of the number is DUALBOOT_BK_SRAM_OFFSET in hack/bt/n64/src/dualboot.h of the
   BKBT repository; nothing compares the two, so change both. The first
   assertion stops a smaller number, which would put this save on top of Tooie's. The others keep the
   0x4A0 bytes inside the SRAM and on a multiple of 8, the size of one EEPROM block; src/core1/eeprom.c
   stops a build whose records add up to another size, so that this number stays true. */
bkbt_sram_offset = 0x7000;
ASSERT(bkbt_sram_offset >= 0x7000, "BKBT: bkbt_sram_offset is below 0x7000, inside the Tooie hack's part of the SRAM");
ASSERT(bkbt_sram_offset % 8 == 0, "BKBT: bkbt_sram_offset is not a multiple of 8");
ASSERT(bkbt_sram_offset + 0x4A0 <= 0x8000, "BKBT: bkbt_sram_offset does not leave 0x4A0 bytes before the end of the SRAM");

/* overlay_load (src/core1/overlay.c) clears an overlay's bss and then writes four checksum words at its
   start, where gChecksums<overlay> is. A file whose .bss is linked ahead of checksums.c.o puts its first
   uninitialised global there instead: the loader overwrites it and nothing warns. Stop the link. */

ASSERT(gChecksumsCC == CC_BSS_START, "BKBT: gChecksumsCC is not at CC_BSS_START (a new .bss is linked ahead of checksums.c.o)");
ASSERT(gChecksumsGV == GV_BSS_START, "BKBT: gChecksumsGV is not at GV_BSS_START (a new .bss is linked ahead of checksums.c.o)");
ASSERT(gChecksumsMMM == MMM_BSS_START, "BKBT: gChecksumsMMM is not at MMM_BSS_START (a new .bss is linked ahead of checksums.c.o)");
ASSERT(gChecksumsTTC == TTC_BSS_START, "BKBT: gChecksumsTTC is not at TTC_BSS_START (a new .bss is linked ahead of checksums.c.o)");

ASSERT(gChecksumsFP == FP_BSS_START, "BKBT: gChecksumsFP is not at FP_BSS_START (a new .bss is linked ahead of checksums.c.o)");
ASSERT(gChecksumsSM == SM_BSS_START, "BKBT: gChecksumsSM is not at SM_BSS_START (a new .bss is linked ahead of checksums.c.o)");

/* Every level overlay is loaded at the end of core2 and has to end below the Stop 'n' Swop block that
   core1 keeps at 0x803A5C00 (src/core1/code_1D5D0.c:127). Past it the game overwrites the overlay's last
   variables, and neither the linker nor an emulator notices. Stop the link. */
BKBT_OVERLAY_CEILING = 0x803A5C00;
ASSERT(core2_VRAM_END <= BKBT_OVERLAY_CEILING, "BKBT: core2 ends above 0x803A5C00: core1 + core2 have grown past the free RAM");
ASSERT(CC_VRAM_END <= BKBT_OVERLAY_CEILING, "BKBT: CC ends above 0x803A5C00: core1 + core2 + CC have grown past the free RAM");
ASSERT(GV_VRAM_END <= BKBT_OVERLAY_CEILING, "BKBT: GV ends above 0x803A5C00: core1 + core2 + GV have grown past the free RAM");

/* The entry stub (asm/entry.s, taken from the ROM, not built from source) clears the boot segment's bss
   with a literal length. A new uninitialised global in src/boot would lie beyond it: never cleared on a
   console, and an emulator, which starts with zeroed RAM, would not show it. Stop the link. */
ASSERT(boot_bk_boot_BSS_SIZE == 0x4570, "BKBT: the boot segment's bss is no longer 0x4570 bytes, which is all the entry stub clears");

/* The compressor splits core1 into its text and data files at core1_DATA_START_OFFSET, and the loader puts
   the data at the next multiple of 16 after the text (src/core1/code_660.c, _rarezip_uncompress). The two
   agree only while core1's text ends on a 16-byte boundary. Stop the link. */
ASSERT(core1_DATA_START_OFFSET == core1_TEXT_END, "BKBT: core1's text does not end on a 16-byte boundary: the loader would put core1's data in the wrong place");
